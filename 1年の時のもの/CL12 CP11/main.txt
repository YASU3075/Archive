#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <math.h>

//-----------------------------------------------
//マクロ定義
//-----------------------------------------------
#define HIGHT_MAX	(25)
#define WIDTH_MAX	(80)

//-----------------------------------------------
//グローバル変数
//-----------------------------------------------
char map[HIGHT_MAX][WIDTH_MAX];


//-----------------------------------------------
//main関数
//-----------------------------------------------
int main(void)
{
	//変数宣言
	int hight, width, hankei;	//円のステータス用
	int y, x;					//円の描画用
	double th, dth;				//角度
	char C[128];				//入力バッファ
	
	//-------------------------------------------
	//入力処理
	printf("******\n");
	printf("縦：");
	scanf("%s", C);
	hight = atoi(C);
	
	printf("横：");
	scanf("%s", C);
	width = atoi(C);
	
	printf("半径：");
	scanf("%s", C);
	hankei = atoi(C);
	//----------------------------------------------

	//----------------------------------------------
	//円描画処理
	if (hankei > 0)
	{//半径が０より大きい場合
		dth = 2.0 / hankei;
	}

	else
	{//半径が０以下の場合
		dth = 6.28;
	}
	
	for (y = 0; y < 25; y++)
	{
		//メモリーの確保
		memset(&map[y][0], ' ', width * 2 + hankei * 2);
	}
	
	for (th = 0.0; th < 6.28; th += dth) 
	{
		x = (int)(hankei * 2 * cos(th) + width * 2);
		y = (int)(hankei * sin(th) + hight);
		
		if (x < 0 || x>79)
		{//xが０より小さいまたは79より大きい場合
			continue;
		}
		
		if (y < 0 || y>23)
		{//yが０より小さいまたは２３より大きい場合
			continue;
		}
		
		map[y][x] = '*';
	}
	
	
	
	for (y = 23; y >= 0; y--) 
	{
		printf("%s\n", &map[y][0]);
	}
	//-------------------------------------------
	
	fflush(stdout); 

	//入力待ち
	rewind(stdin);
	getchar(); 
	
	return 0;
}
